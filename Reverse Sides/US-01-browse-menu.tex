\begin{frame}{Browse Restaurant Menu}

\noindent\textbf{Role:} Customer\\
\textbf{Priority:} Must Have

\textbf{User Story:} As a customer, I want to browse the restaurant menu, so that I can see available dishes and their prices.

\textbf{Functional Acceptance Criteria}
\begin{itemize}
  \item Can the customer view the restaurant menu without placing an order?
  \item Does the menu display the name and price of each available dish?
  \item Are unavailable dishes excluded from the displayed menu?
  \item Does the system display updated prices after the restaurant manager modifies them?
  \item Does the system display an appropriate message when no dishes are available?
\end{itemize}

\end{frame}

\begin{frame}{Browse Restaurant Menu}

\textbf{Non-Functional Requirements}
\begin{itemize}
  \item \textbf{Performance:} The menu should load within 2 seconds for at least 95\% of requests under normal operating conditions.
  \item \textbf{Usability:} Dish names and prices must be clearly readable on desktop and mobile devices.
  \item \textbf{Reliability:} The application must not display incomplete menu data when retrieval fails.
\end{itemize}

\textbf{Notes and Edge Cases}
\begin{itemize}
  \item The restaurant menu is empty.
  \item A dish becomes unavailable while the customer is browsing.
  \item The restaurant manager updates a dish price.
  \item The menu service is temporarily unavailable.
\end{itemize}

\end{frame}
