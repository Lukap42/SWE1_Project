\begin{frame}{View Upcoming Reservations}

\noindent\textbf{Role:} Restaurant Manager\\
\textbf{Priority:} Must Have

\textbf{User Story:} As a restaurant manager, I want to view all upcoming reservations, so that I can manage table availability.

\textbf{Functional Acceptance Criteria}
\begin{itemize}
  \item Can the restaurant manager view a list of upcoming reservations?
  \item Does each reservation display its date and time?
  \item Does the system display only reservations that have not yet reached their scheduled time?
  \item Can the manager identify the reserved table for each reservation?
  \item Does the system display an appropriate message when no upcoming reservations exist?
  \item Are users without the required permissions prevented from accessing reservation information?
\end{itemize}

\end{frame}

\begin{frame}{View Upcoming Reservations}

\textbf{Non-Functional Requirements}
\begin{itemize}
  \item \textbf{Performance:} The reservation list should load within 2 seconds for at least 95\% of requests containing up to 500 reservations.
  \item \textbf{Security:} Only authorized restaurant personnel may access reservation data.
  \item \textbf{Reliability:} Confirmed reservations must remain available after a system restart.
  \item \textbf{Usability:} Reservations should be displayed in chronological order.
\end{itemize}

\textbf{Notes and Edge Cases}
\begin{itemize}
  \item There are no upcoming reservations.
  \item Multiple reservations exist for the same time.
  \item A reservation is cancelled while the manager is viewing the list.
  \item The reservation database is temporarily unavailable.
\end{itemize}

\end{frame}
