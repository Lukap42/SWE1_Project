\begin{frame}{User: Customer}
Must have:
\begin{itemize}
\item As a customer, I want to browse the restaurant menu, so that I can see available dishes and their prices.
\item As a customer, I want to order food, so that I can have meals delivered to my chosen location.
\item As a customer, I want to make a table reservation online, so that I can secure a table at preferred time.
\end{itemize}
Should have:
\begin{itemize}
\item As a customer, I want to create an account, so that I can manage my reservations and orders.
\item As a customer, I want to track my order status, so that I know when my food will be delivered.
\end{itemize}
\end{frame}

\begin{frame}{User: Customer}
Could have:
\begin{itemize}
\item As a customer, I want to view my previous orders, so that I can easily reorder my favorite meals.
\end{itemize}
Won't have:
\begin{itemize}
\item As a customer, I want to rate dishes and deliveries, so that I can share my experience with the restaurant.
\end{itemize}
\end{frame}

\begin{frame}{User: Customer}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Must & As a customer, I want to browse the restaurant menu, so that I can see available dishes and their prices.
     & Application menu shows dish name and price for available items. Unavailable items are not shown. \\
\hline
Must & As a customer, I want to order food, so that I can have meals delivered to my chosen location.
     & Orders can be submitted with delivery address and contact information. System validates payment and sends confirmation. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Customer}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Must & As a customer, I want to make a table reservation online, so that I can secure a table at preferred time.
     & User submits contact information and system validates if reservation is available. On success, sends confirmation. \\
\hline
Should & As a customer, I want to create an account, so that I can manage my reservations and orders.
       & Registration saves account information and creates unique identifier. Credentials stored hashed. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Customer}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Should & As a customer, I want to track my order status, so that I know when my food will be delivered.
       & Application shows order contents along with estimated delivery time. \\
\hline
Could & As a customer, I want to view my previous orders, so that I can easily reorder my favorite meals.
      & History is available only to users with an account. Displays order contents, price and date of order. \\
\hline
Won't & As a customer, I want to rate dishes and deliveries, so that I can share my experience with the restaurant.
      & Out of scope for the application. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Restaurant manager}
Must have:
\begin{itemize}
\item As a restaurant manager, I want to view all upcoming reservations, so that I can manage table availability.
\item As a restaurant manager, I want to assign delivery orders to available drivers, so that deliveries can be completed efficiently.
\item As a restaurant manager, I want different user roles to have different access permissions, so that sensitive business information remains protected.
\end{itemize}
\end{frame}

\begin{frame}{User: Restaurant manager}
Should have:
\begin{itemize}
\item As a restaurant manager, I want to view order statistics, so that I can identify the most popular dishes.
\item As a restaurant manager, I want the system to prevent double bookings, so that the same table cannot be reserved by multiple customers at the same time.
\item As a restaurant manager, I want to update menu items and prices, so that customers always see the current restaurant offer.
\end{itemize}
\end{frame}

\begin{frame}{User: Restaurant manager}
Could have:
\begin{itemize}
\item As a restaurant manager, I want to analyze reservation trends, so that I can better prepare for busy periods.
\end{itemize}
Won't have:
\begin{itemize}
\item As a restaurant manager, I want to predict future sales using historical data, so that I can optimize inventory and staffing.
\end{itemize}
\end{frame}

\begin{frame}{User: Restaurant manager}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Must & As a restaurant manager, I want to view all upcoming reservations, so that I can manage table availability.
     & Application shows all booked reservations in a list with time and date visible. \\
\hline
Must & As a restaurant manager, I want to assign delivery orders to available drivers, so that deliveries can be completed efficiently.
     & When assigning a delivery, application shows each available driver along with number of already planned deliveries for them. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Restaurant manager}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Must & As a restaurant manager, I want different user roles to have different access permissions, so that sensitive business information remains protected.
     & When an employee is assigned a role, they are unable to view information tagged for other roles \\
\hline
Should & As a restaurant manager, I want to view order statistics, so that I can identify the most popular dishes.
       & Application allows a privileged user to view sales of food items ordered by number of sales along with their count. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Restaurant manager}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Should & As a restaurant manager, I want the system to prevent double bookings, so that the same table cannot be reserved by multiple customers at the same time.
       & When two users try to book the same table at the same time up to the minute, only one of them is successfully processed by the system. \\
\hline
Should & As a restaurant manager, I want to update menu items and prices, so that customers always see the current restaurant offer.
       & Application allows a privileged user to modify live prices of menu items and add or remove items on offer. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Restaurant manager}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Could & As a restaurant manager, I want to analyze reservation trends, so that I can better prepare for busy periods.
      & Given a table is regularily reserved, application warns a privileged user which tables are highly contested. \\
\hline
Won't & As a restaurant manager, I want to predict future sales using historical data, so that I can optimize inventory and staffing.
      & Won't be implemented. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Delivery driver}
Must have:
\begin{itemize}
\item As a delivery driver, I want to view my assigned deliveries, so that I know which orders I need to deliver and where.
\end{itemize}
Should have:
\begin{itemize}
\item As a delivery driver, I want to mark an order as delivered, so that the system can update its status automatically.
\end{itemize}
Could have:
\begin{itemize}
\item As a delivery driver, I want to view the estimated delivery route, so that I can plan my journey and deliver orders efficiently.
\end{itemize}
Won't have:
\begin{itemize}
\item As a delivery driver, I want to communicate with customers through the application, so that I can inform them about delivery delays or special circumstances.
\end{itemize}
\end{frame}

\begin{frame}{User: Delivery driver}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Must & As a delivery driver, I want to view my assigned deliveries, so that I know which orders I need to deliver and where.
     & Application shows list of deliveries currently assigned to the driver. \\
\hline
Should & As a delivery driver, I want to mark an order as delivered, so that the system can update its status automatically.
       & Application allows the marking a delivery as completed, removing it from the list of active deliveries. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Delivery driver}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Could & As a delivery driver, I want to view the estimated delivery route, so that I can plan my journey and deliver orders efficiently.
      & Application displays the delivery address and a suggested route for each assigned delivery. \\
\hline
Won't & As a delivery driver, I want to communicate with customers through the application, so that I can inform them about delivery delays or special circumstances.
      & Out of scope for the application. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Restaurant owner}
Must have:
\begin{itemize}
\item As a restaurant owner, I want to view financial reports, so that I can monitor the restaurant's revenue and financial performance.
\end{itemize}
Should have:
\begin{itemize}
\item As a restaurant owner, I want to compare financial performance across different months, so that I can identify changes in revenue and costs.
\end{itemize}
\end{frame}

\begin{frame}{User: Restaurant owner}
Could have:
\begin{itemize}
\item As a restaurant owner, I want to export financial reports, so that I can use the data for accounting and record keeping.
\end{itemize}
Won't have:
\begin{itemize}
\item As a restaurant owner, I want to receive automatic notifications when the restaurant receives a new order, so that I can stay informed about daily operations without checking the application.
\end{itemize}
\end{frame}

\begin{frame}{User: Restaurant owner}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Must & As a restaurant owner, I want to view financial reports, so that I can monitor the restaurant's revenue and financial performance.
     & Application shows breakdown of financial earnings and costs categorized for a given month. \\
\hline
Should & As a restaurant owner, I want to compare financial performance across different months, so that I can identify changes in revenue and costs.
       & Application allows the owner to select two or more months and compare their total revenue and costs. \\
\hline
\end{tabular}
\end{frame}

\begin{frame}{User: Restaurant owner}
\begin{tabular}{| l | p{4cm} | p{4cm} |}
\hline
Priority & User Story & Acceptability criteria \\
\hline
Could & As a restaurant owner, I want to export financial reports, so that I can use the data for accounting and record keeping.
      & Application allows the owner to download a financial report in CSV or PDF format for a selected month. \\
\hline
Won't & As a restaurant owner, I want to receive automatic notifications when the restaurant receives a new order, so that I can stay informed about daily operations without checking the application.
      & Out of scope for the application. \\
\hline
\end{tabular}
\end{frame}
